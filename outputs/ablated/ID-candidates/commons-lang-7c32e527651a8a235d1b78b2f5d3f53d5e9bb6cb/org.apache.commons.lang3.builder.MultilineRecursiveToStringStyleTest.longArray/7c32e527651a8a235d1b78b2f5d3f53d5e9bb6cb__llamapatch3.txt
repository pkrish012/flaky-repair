class DeterministicMultilineRecursiveToStringStyle extends MultilineRecursiveToStringStyle {

    @Override
    protected void appendDetail(StringBuffer buffer, String fieldName, Object value) throws Exception {
        if (value instanceof Object[]) {
            Object[] array = (Object[]) value;
            Arrays.sort(array);
            super.appendDetail(buffer, fieldName, array);
        } else {
            super.appendDetail(buffer, fieldName, value);
        }
    }

    // Usage example within the test class
    private String toString(final Object object) {
        return new ReflectionToStringBuilder(object, new DeterministicMultilineRecursiveToStringStyle()).toString();
    }
}

class OrderedFieldsMultilineRecursiveToStringStyle extends MultilineRecursiveToStringStyle {

    @Override
    public void appendFields(Buffer buffer, Object object) {
        Class<?> clazz = object.getClass();
        Field[] fields = clazz.getDeclaredFields();

        // Sort fields by name to ensure consistent order
        Arrays.sort(fields, Comparator.comparing(Field::getName));

        for (Field field : fields) {
            try {
                field.setAccessible(true); // Ensure we can access private fields
                appendField(buffer, field.getName(), field.get(object));
            } catch (Exception e) {
                throw new RuntimeException(e);
            }
        }
    }
