            FieldUtils.getFields(value.getClass())

                            builder.append(field.getName(), field.get(value));
                        } catch (IllegalAccessException e) {

                    });

private String toString(final Object object) {
    return new ReflectionToStringBuilder(object, new MultilineRecursiveToStringStyle()) {
        @Override
        protected void appendFieldsInDepthFirst(ToStringBuilder builder, T value) {
            Arrays.stream(FieldUtils.getAllFieldsList(value.getClass()))
               .filter(f ->!f.getName().equals("this$0"))
               .sorted(Comparator.comparing(Field::getName))
               .forEachOrdered(f -> {
                    f.setAccessible(true);
                    try {
                        builder.append(f.getName(), f.get(value));
                    } catch (IllegalAccessException ex) {
                        throw new InternalError(ex);
                    }
                });
        }
    }.toString();
}
