def get_sorted_fields(cls):
    """Return fields sorted alphabetically."""
    import inspect
    
    # Get all members of the class
    members = inspect.getmembers(cls)
    
    # Filter out non-field members
    fields = [name for name, _ in members if isinstance(_, type(lambda: None))]
    
    # Sort fields alphabetically
    sorted_fields = sorted(fields)
    
    return sorted_fields


class MultilineRecursiveToStringStyleFixed(MultilineRecursiveToStringStyle):
    def __init__(self):
        super().__init__()
        
    def appendFields(self, obj, buf):
        cls = obj.__class__
        sorted_field_names = get_sorted_fields(cls)
        
        for field_name in sorted_field_names:
            field_value = getattr(obj, field_name)
            
            if field_value is not None:
                buf.append(field_name + "=" + str(field_value) + ", ")
                
        # Remove trailing comma and space
        if buf.toString().endswith(', '):
            buf.removeSuffix(', ')
    

ReflectionToStringBuilder.setDefaultStyle(MultilineRecursiveToStringStyleFixed)
