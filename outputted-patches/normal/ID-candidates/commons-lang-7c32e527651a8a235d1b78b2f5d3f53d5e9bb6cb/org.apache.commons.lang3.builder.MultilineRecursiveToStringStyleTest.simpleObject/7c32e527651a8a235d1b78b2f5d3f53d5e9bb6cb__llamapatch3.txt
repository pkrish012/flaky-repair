class ToStringBuilder {
    private final Object object;

    public ToStringBuilder(Object object) {
        this.object = object;
    }

    public String toString() {
        StringBuffer buffer = new StringBuffer(getClassPrefix(object)).append('[');
        for (java.lang.reflect.Field field : object.getClass().getDeclaredFields()) {
            field.setAccessible(true);
            append(buffer, field.getName(), field.get(object), false);
        }
        buffer.deleteCharAt(buffer.length()-1); // Remove last comma
        buffer.append(']');
        return buffer.toString();
    }

    private void append(StringBuffer buffer, String fieldName, Object value, boolean alreadyThere) {
        if (alreadyThere) {
            return;
        }
        buffer.append(System.lineSeparator() + "  ");
        buffer.append(fieldName);
        buffer.append('=');
        buffer.append(value);
        buffer.append(',');
    }

    private String getClassPrefix(final Object object) {
        return object.getClass().getName() + "@" + Integer.toHexString(System.identityHashCode(object));
