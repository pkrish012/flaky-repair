class MultilineRecursiveToStringStyle extends ToStringStyle {
    public void append(StringBuffer buffer, String fieldName, Object value, boolean alreadyThere) {
        if (alreadyThere) {
            super.append(buffer, fieldName, value, alreadyThere);
            return;
        }
        buffer.append(System.lineSeparator())
               .append("  ")
               .append(fieldName)
               .append("=")
               .append(String.valueOf(value))
               .append(",");
    }

    public void appendEnd(StringBuffer buffer, String objectToString) {
        // Remove trailing comma
        int length = buffer.length();
        if (length > 0 && buffer.charAt(length - 1) == ',') {
            buffer.setLength(length - 1);
        }
        buffer.append(']');
    }
}
class ToStringBuilder {
    private final StringBuffer buffer = new StringBuffer();

    public ToStringBuilder(Object object) {
        buffer.append(getClassPrefix(object)).append("[");
        for (java.lang.reflect.Field field : object.getClass().getDeclaredFields()) {
            field.setAccessible(true);
            appendField(field.getName(), field.get(object));
        }
    }

    private void appendField(String fieldName, Object value) {
        buffer.append(System.lineSeparator())
               .append("  ")
               .append(fieldName)
               .append("=")
               .append(String.valueOf(value))
               .append(",");
    }

    public String toString() {
        buffer.append(']');
        return buffer.toString();
    }
}
